package proxy

import (
	"context"
	"net/http"
	"net/http/httptest"
	"sync"
	"sync/atomic"
	"testing"
	"time"

	"go-reauth-proxy/pkg/grpc/pb"
	"go-reauth-proxy/pkg/models"
)

// These barriers model an authorization decision made before logout whose RPC
// response arrives afterward. No listener or timing-dependent backend is used.
func TestAuthCacheInvalidationSeparatesPendingAuthorization(t *testing.T) {
	for _, oldAllowed := range []bool{true, false} {
		for _, mode := range []string{"legacy_verify", "combined_verify", "legacy_preflight", "combined_preflight", "combined"} {
			for _, invalidation := range []string{"identity", "clear", "logout_cookie", "ttl_reconfigure"} {
				for _, order := range []string{"late_fill", "new_request_before_old_returns"} {
					decisionName := "old_deny"
					if oldAllowed {
						decisionName = "old_allow"
					}
					t.Run(decisionName+"/"+mode+"/"+invalidation+"/"+order, func(t *testing.T) {
						h := &Handler{authCache: newAuthStateCache(), preflightCache: newPreflightStateCache()}
						cfg := models.AuthConfig{AuthURL: "/verify", AuthCacheTTL: 60, AuthCacheFailTTL: 60}
						if invalidation == "ttl_reconfigure" {
							h, _ = newAdditionalProxyTestHandler(t)
						}
						started, freshStarted, release := make(chan struct{}), make(chan struct{}), make(chan struct{})
						var releaseOnce sync.Once
						unblock := func() { releaseOnce.Do(func() { close(release) }) }
						t.Cleanup(unblock)
						var calls atomic.Int32
						oldDecision := func() bool {
							n := calls.Add(1)
							if n == 1 {
								close(started)
								<-release
								return oldAllowed
							}
							if n == 2 {
								close(freshStarted)
							}
							return !oldAllowed
						}
						verify := func(allowed bool) *pb.VerifyAuthResponse {
							if allowed {
								return &pb.VerifyAuthResponse{Success: true, Status: http.StatusOK, LoginAuthenticated: true}
							}
							return &pb.VerifyAuthResponse{Status: http.StatusUnauthorized}
						}
						h.authBridge = testAuthBridge{
							supports: mode != "legacy_verify" && mode != "legacy_preflight",
							verify: func(context.Context, *pb.VerifyAuthRequest) (*pb.VerifyAuthResponse, error) {
								return verify(oldDecision()), nil
							},
							preflight: func(context.Context, *pb.PreflightAuthRequest) (*pb.PreflightAuthResponse, error) {
								return &pb.PreflightAuthResponse{Deny: !oldDecision()}, nil
							},
							authorize: func(context.Context, *pb.AuthorizeHttpRequest) (*pb.AuthorizeHttpResponse, error) {
								allowed := oldDecision()
								return &pb.AuthorizeHttpResponse{
									Preflight: &pb.PreflightAuthResponse{Deny: !allowed}, Verify: verify(allowed),
									PreflightCacheScope: pb.AuthCacheScope_AUTH_CACHE_SCOPE_EXACT_REQUEST,
									VerifyCacheScope:    pb.AuthCacheScope_AUTH_CACHE_SCOPE_HOST,
								}, nil
							},
						}
						request := func() *http.Request {
							r := httptest.NewRequest(http.MethodGet, "https://app.example/private", nil)
							r.AddCookie(&http.Cookie{Name: authSessionCookieName, Value: "revoked-session"})
							return r
						}
						invoke := func() bool {
							r := request()
							auth := newRequestAuthContext(r, "192.0.2.1", "login_first", routedBackend{})
							switch mode {
							case "legacy_preflight", "combined_preflight":
								return !preflightStopsHTTPAuthorization(h.runPreflight(r, cfg, "192.0.2.1", true, "login_first", "", auth))
							case "combined":
								e, used := h.executeCombinedHTTPAuth(r, cfg, "192.0.2.1", "login_first", true, "", auth)
								if !used || preflightStopsHTTPAuthorization(e.preflight) {
									return false
								}
								if e.auth.entry != nil {
									return e.auth.entry.result.allowed
								}
								return e.auth.plan.result.allowed
							default:
								e := h.executeAuthCheck(r, cfg, "192.0.2.1", "login_first", "", auth)
								if e.entry != nil {
									return e.entry.result.allowed
								}
								return e.plan.result.allowed
							}
						}
						oldDone := make(chan bool, 1)
						go func() { oldDone <- invoke() }()
						select {
						case <-started:
						case <-time.After(time.Second):
							t.Fatal("old RPC did not start")
						}
						switch invalidation {
						case "identity":
							h.authCacheInvalidateByIdentityKeys(requestIdentityKey(request(), "192.0.2.1"))
						case "clear":
							h.clearAuthCache()
						case "logout_cookie":
							h.authCacheInvalidateForSetCookieMutation(request(), "192.0.2.1", []string{authSessionCookieName + "=; Max-Age=0; Path=/"})
						case "ttl_reconfigure":
							disabled := cfg
							disabled.AuthCacheTTL, disabled.AuthCacheFailTTL = 0, 0
							if err := h.SetAuthConfig(disabled); err != nil {
								t.Fatal(err)
							}
							if err := h.SetAuthConfig(cfg); err != nil {
								t.Fatal(err)
							}
						}
						if order == "new_request_before_old_returns" {
							freshDone := make(chan bool, 1)
							go func() { freshDone <- invoke() }()
							select {
							case <-freshStarted:
								if got := <-freshDone; got == oldAllowed {
									t.Error("post-invalidation request received the old decision")
								}
							case <-time.After(time.Second):
								t.Error("post-invalidation request joined the old singleflight")
								unblock()
								<-freshDone
							}
							unblock()
						} else {
							unblock()
						}
						if got := <-oldDone; got != oldAllowed {
							t.Error("invalidation changed the original in-flight result")
						}
						if got := invoke(); got == oldAllowed {
							t.Error("old RPC republished its decision after invalidation")
						}
						if calls.Load() < 2 {
							t.Error("post-invalidation authorization was not rechecked")
						}
					})
				}
			}
		}
	}
}
